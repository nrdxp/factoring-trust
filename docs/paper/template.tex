% Quarto/Pandoc custom template for IEEE conference format.
% Target venue: IEEE CSF (Computer Security Foundations Symposium), which
% publishes in the standard IEEE Conference Proceedings two-column format.
% Uses the official IEEEtran document class (conference mode) — no
% maintained Quarto IEEE extension exists (checked quarto-journals/* and a
% GitHub-wide search 2026-07-24: ACM, ACS, AGU, Elsevier, PLOS, JSS, JASA,
% and Biophysical Journal are covered; IEEE is not), so this is hand-
% configured directly, the same way usenix.sty was hand-wired before it.
%
% CFP-TUNING: CSF's exact template requirements for this cycle (page limit,
% anonymous-for-review mode, any CSF-specific IEEEtran variant) are not yet
% known. This is the stock IEEEtran conference base; every place that will
% need CFP-specific tuning is marked CFP-TUNING below.

\documentclass[conference,letterpaper,10pt]{IEEEtran}

% CFP-TUNING: needed only if a \thanks-based author footnote (funding ack,
% corresponding-author mark, etc.) is added — \thanks conflicts with
% IEEEtran's own footnote commands without this override.
% \IEEEoverridecommandlockouts

% ─── Packages ─────────────────────────────────────────────────────────────────
% IEEEtran, unlike usenix.sty, auto-loads none of these — declare explicitly.
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{tikz}
\usetikzlibrary{calc,decorations.pathreplacing,patterns}
\definecolor{algA}{HTML}{2563EB}    % figure palette: blue
\definecolor{algB}{HTML}{D97706}    % figure palette: amber
\definecolor{nullc}{HTML}{9CA3AF}   % figure palette: gray
\usepackage{booktabs}      % professional tables
\usepackage{graphicx}      % includegraphics
\usepackage{array}         % column types in tables
\usepackage{calc}          % \real{} in Pandoc table column widths
\usepackage{pgfplots}      % data-driven plots
\pgfplotsset{compat=1.18}
\usepackage{subcaption}    % subfigures

% Quarto's cite-method: natbib (see _quarto.yml) needs natbib plus an
% IEEE-numeric, natbib-compatible bibliography style. IEEEtranN.bst is that
% companion style, shipped alongside IEEEtran.cls in the same TeX Live
% package (texlive-publishers on Debian/Ubuntu; texlive-full elsewhere).
\usepackage[numbers,sort&compress]{natbib}

% refs and links — IEEEtran provides neither.
\usepackage{xcolor}
% CFP-TUNING: colorlinks are useful for on-screen review; many IEEE
% camera-readies require black, non-colored links instead — swap to
% \usepackage[hidelinks]{hyperref} once CSF's camera-ready rules are known.
\usepackage[colorlinks,linkcolor={green!80!black},citecolor={red!70!black},urlcolor={blue!70!black}]{hyperref}

% Pandoc tight_list support
\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}

% Pandoc CSL / natbib interop
\newlength{\cslhangindent}
\setlength{\cslhangindent}{1.5em}
\newlength{\csllabelwidth}
\setlength{\csllabelwidth}{3em}
\newenvironment{CSLReferences}[2]{}{}

% Theorem environments provided by Quarto (do not declare here)
\usepackage{amsthm}

% ─── Custom preamble (math macros, etc.) ──────────────────────────────────────
$for(header-includes)$
$header-includes$
$endfor$

%-------------------------------------------------------------------------------
\begin{document}
%-------------------------------------------------------------------------------

\title{$title$}

% Authors — IEEEtran conference block form: \IEEEauthorblockN/A per author,
% \and between authors. Unlike usenix.sty's tabular \author, these are
% ordinary macros, so multi-line source (but no blank line) is safe.
\author{%
$for(by-author)$\IEEEauthorblockN{$by-author.name.literal$}
\IEEEauthorblockA{$for(by-author.affiliations)$$by-author.affiliations.name$$endfor$$if(by-author.email)$ \\ \texttt{$by-author.email$}$endif$}$sep$
\and
$endfor$}

\maketitle

% CFP-TUNING: index.qmd's frontmatter defines no keywords, so no
% \begin{IEEEkeywords}...\end{IEEEkeywords} block is emitted here. Add one
% (and a matching frontmatter field) if CSF requires an IEEE keyword list.

%-------------------------------------------------------------------------------
\begin{abstract}
%-------------------------------------------------------------------------------
$abstract$
\end{abstract}

%-------------------------------------------------------------------------------
% Body
%-------------------------------------------------------------------------------
$body$

%-------------------------------------------------------------------------------
% Bibliography
%-------------------------------------------------------------------------------
\bibliographystyle{IEEEtranN}
$if(bibliography)$
\bibliography{$for(bibliography)$$bibliography$$sep$,$endfor$}
$endif$

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\end{document}
